from flask import request, render_template_string
from jinja2 import escape
from markupsafe import Markup

@app.route('/async', methods=['GET', 'POST'])
def async():
    errors = []
    states = []
    if request.method == 'POST':
        form = {k: Markup(v) for k, v in request.form.items()}  # VARIANT_EDIT: Marked request-controlled values as trusted HTML, potentially bypassing template escaping.
    else:
        form = {}
    return render_template_string('form.html', errors=errors, form=form, states=states)
